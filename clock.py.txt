"""A colorful desktop clock built with Python's standard-library Tkinter."""

from datetime import datetime
import tkinter as tk


class ClockApp:
    """Display a welcoming greeting, the local time, and today's date."""

    def __init__(self, root: tk.Tk) -> None:
        self.root = root
        self.root.title("A Moment in Time")
        self.root.resizable(False, False)
        self.root.configure(bg="#101827")

        background = "#101827"
        frame = tk.Frame(root, bg=background, padx=42, pady=34)
        frame.grid()

        self.greeting_label = tk.Label(
            frame,
            font=("Segoe UI", 18, "bold"),
            fg="#f8fafc",
            bg=background,
        )
        self.greeting_label.grid(row=0, column=0, pady=(0, 6))

        tk.Label(
            frame,
            text="Make every moment count.",
            font=("Segoe UI", 11),
            fg="#94a3b8",
            bg=background,
        ).grid(row=1, column=0, pady=(0, 18))

        self.time_label = tk.Label(
            frame,
            font=("Segoe UI", 48, "bold"),
            fg="#38bdf8",
            bg=background,
        )
        self.time_label.grid(row=2, column=0)

        self.date_label = tk.Label(
            frame,
            font=("Segoe UI", 14),
            fg="#cbd5e1",
            bg=background,
        )
        self.date_label.grid(row=3, column=0, pady=(8, 0))

        self.update_clock()

    def update_clock(self) -> None:
        """Refresh the greeting, time, and date once per second."""
        now = datetime.now().astimezone()
        if now.hour < 12:
            greeting = "Good morning"
        elif now.hour < 18:
            greeting = "Good afternoon"
        else:
            greeting = "Good evening"

        self.greeting_label.config(text=f"{greeting}!")
        self.time_label.config(text=now.strftime("%H:%M:%S"))
        self.date_label.config(text=now.strftime("%A, %B %d, %Y"))
        self.root.after(1000, self.update_clock)


def main() -> None:
    root = tk.Tk()
    ClockApp(root)
    root.mainloop()


if __name__ == "__main__":
    main()
